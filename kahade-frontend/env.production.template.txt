# ============================================================================

# ============================================================================
# INSTRUCTIONS:
# 1. Copy this file to kahade-frontend/.env.production
# 2. Replace YOUR_DOMAIN with your actual domain (e.g., kahade.com)
# 3. Rebuild the frontend: npm run build
# ============================================================================

# API Configuration
# Replace YOUR_DOMAIN with your actual domain
VITE_API_URL=https://api.YOUR_DOMAIN/api/v1

# Application Mode (landing | app | admin)
# This will be set during build for each subdomain
VITE_APP_MODE=landing

# Domain Configuration
# Replace YOUR_DOMAIN with your actual domain
VITE_BASE_DOMAIN=YOUR_DOMAIN
VITE_LANDING_URL=https://YOUR_DOMAIN
VITE_APP_URL=https://app.YOUR_DOMAIN
VITE_ADMIN_URL=https://admin.YOUR_DOMAIN

# Cookie Configuration
# Replace YOUR_DOMAIN with your actual domain (note the leading dot)
VITE_COOKIE_DOMAIN=.YOUR_DOMAIN

# Feature Flags
VITE_ENABLE_DEBUG=false
VITE_ENABLE_ANALYTICS=true
